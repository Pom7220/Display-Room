// ES5 ONLY — Chrome 42 / Android 4.2 Compatible

function fetchCalendarForRoom(email, tabletKey, startISO, endISO) {

  var workerOrigin = window.location.origin;

  var url = workerOrigin + '/api/calendar'

    + '?room=' + encodeURIComponent(email)

    + '&startDateTime=' + startISO

    + '&endDateTime=' + endISO;

  return fetch(url, { 

    headers: { 

      'X-Tablet-Key': tabletKey 

    } 

  })

    .then(function(r) {

      if (!r.ok) return { value: [] };

      return r.json();

    })

    .then(function(d) {

      return filterMeetings(d.value || []);

    })

    .catch(function() { 

      return []; 

    });

}

// Configuration state setup

var cfg = {

  roomemail: localStorage.getItem('roomdisplay_email'),

  roomname: localStorage.getItem('roomdisplay_name'),

  tabletkey: localStorage.getItem('roomdisplay_tabletkey') || 'YOUR_STATIC_TABLET_SECRET'

};

function refreshData() {

  if (!cfg.roomemail || !cfg.tabletkey) {

    showConfig();

    return;

  }

  var start = new Date();

  start.setHours(0, 0, 0, 0);

  var end = new Date();

  end.setDate(end.getDate() + 5);

  dbg('Fetching data from Cloudflare Worker Proxy for ' + cfg.roomname);

  fetchCalendarForRoom(cfg.roomemail, cfg.tabletkey, start.toISOString(), end.toISOString())

    .then(function(meetings) {

      // Direct pass-through to your existing layout rendering functions

      renderUi(meetings);

      sendHeartbeat('OK', 'Synced successfully via Worker proxy');

    })

    .catch(function(err) {

      dbg('Error loading calendar content: ' + err.message);

      sendHeartbeat('ERROR', 'Proxy fetching failure: ' + err.message);

    });

}
